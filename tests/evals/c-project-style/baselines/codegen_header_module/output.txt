/* Opaque DNS resolver handle. Allocated via xylem_dns_create. */
typedef struct xylem_dns_s xylem_dns_t;

/**
 * @brief Create a DNS resolver.
 *
 * @return New resolver, or NULL on allocation failure.
 */
extern xylem_dns_t* xylem_dns_create(void);

/**
 * @brief Destroy a DNS resolver. Safe to call with NULL.
 *
 * @param dns  Resolver to destroy.
 */
extern void xylem_dns_destroy(xylem_dns_t* dns);

/**
 * @brief Resolve a hostname to an address.
 *
 * @param dns   Resolver handle.
 * @param host  Hostname to resolve.
 *
 * @return 0 on success, -1 on failure.
 */
extern int xylem_dns_resolve(xylem_dns_t* dns, const char* host);
